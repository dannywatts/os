Current goal: UEFI application that prints "Hello, world!"
https://www.rodsbooks.com/efi-programming/hello.html
Fedora packages installed: gcc gnu-efi gnu-efi-devel qemu-kvm edk2-ovmf
gcc -I/usr/include/efi -I/usr/include/efi/x86_64 -I/usr/include/efi/protocol -fno-stack-protector -fpic -fshort-wchar -mno-red-zone -Wall -DEFI_FUNCTION_WRAPPER -c main.c -o main.o
ld -nostdlib -znocombreloc -T /usr/lib/elf_x86_64_efi.lds -shared -Bsymbolic -L /usr/lib /usr/lib/crt0-efi-x86_64.o main.o -o bootx64.so -lefi -lgnuefi
objcopy -j .text -j .sdata -j .data -j .dynamic -j .dynsym -j .rel -j rela -j .reloc -O efi-app-x86_64 bootx64.so bootx64.efi
mkdir drive
mv bootx64.efi drive
qemu-system-x86_64 -bios /usr/share/edk2/ovmf/OVMF_CODE.fd -drive format=raw,file=fat:rw:./drive
